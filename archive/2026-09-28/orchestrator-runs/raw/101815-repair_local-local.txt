"""src/algo.py – Sparse Auditor (Algorithm YB‑0003)

Provides:
* CFFI/ctypes bindings to the compiled core library ``libcore.so``.
* A pure‑Python reference implementation that mirrors the C behaviour
  (used for sanity checks in unit tests).
* The high‑level detector class ``SparseAuditor`` exposing a simple API:

    >>> auditor = SparseAuditor("model.bin", tau=0.55)
    >>> score, flag = auditor.detect(x)

The detector returns the average Gini sparsity over all hidden layers and
a binary OOD/adv flag (1 → out‑of‑distribution) based on a single threshold
``tau``.
"""

from __future__ import annotations

import ctypes as ct
import os
from pathlib import Path
from typing import Tuple, Optional

import numpy as np

# ----------------------------------------------------------------------
# Load the shared library (located next to this file)
# ----------------------------------------------------------------------
_lib_path = Path(__file__).with_name("libcore.so")
if not _lib_path.is_file():
    raise FileNotFoundError(f"Compiled core library not found at {_lib_path}")

_lib = ct.CDLL(str(_lib_path))

# ----------------------------------------------------------------------
# C API signatures (as declared in src/core.h)
# ----------------------------------------------------------------------
_lib.sa_load_model.argtypes = [ct.c_char_p]
_lib.sa_load_model.restype = ct.c_void_p

_lib.sa_free_model.argtypes = [ct.c_void_p]
_lib.sa_free_model.restype = None

_lib.sa_compute_gini.argtypes = [
    ct.c_void_p,
    ct.POINTER(ct.c_float),
    ct.c_int,
    ct.c_int,
    ct.POINTER(ct.c_float)
]
_lib.sa_compute_gini.restype = ct.c_int

_lib.sa_layer_stats.argtypes = [
    ct.c_void_p,
    ct.POINTER(ct.c_float),
    ct.c_int,
    ct.c_int,
    ct.POINTER(ct.c_float),
    ct.POINTER(ct.c_float)
]
_lib.sa_layer_stats.restype = ct.c_int

# ----------------------------------------------------------------------
# Helper: pure‑Python reference implementation
# ----------------------------------------------------------------------
def _gini_numpy(a: np.ndarray) -> float:
    """Gini coefficient for a non‑negative 1‑D array."""
    if a.size <= 1:
        return 0.0
    s = a.sum()
    if s == 0.0:
        return 0.0
    sorted_a = np.sort(a)
    n = a.size
    cum = np.cumsum(sorted_a, dtype=np.float64)
    gini = (n + 1 - 2 * np.sum(cum) / s) / n
    return float(gini)


def _load_weights_from_binary(path: str):
    """Read the binary model file and return a list of weight matrices."""
    with open(path, "rb") as f:
        n_layers = int(np.frombuffer(f.read(4), dtype=np.int32)[0])
        weights = []
        for _ in range(n_layers):
            out_dim = int(np.frombuffer(f.read(4), dtype=np.int32)[0])
            in_dim = int(np.frombuffer(f.read(4), dtype=np.int32)[0])
            w = np.frombuffer(f.read(out_dim * in_dim * 4),
                              dtype=np.float32).reshape(out_dim, in_dim)
            weights.append(w.copy())
    return weights


def _forward_activations(model_path: str, x: np.ndarray):
    """
    Pure‑Python forward pass using the same architecture as the C code.
    Returns hidden activations (post‑ReLU) for all hidden layers,
    excluding the final output layer.
    """
    from src.mlp import MLP

    weights = _load_weights_from_binary(model_path)
    # infer dimensions
    d = weights[0].shape[1]
    h1 = weights[0].shape[0]
    h2 = weights[1].shape[0] if len(weights) > 2 else h1
    c = weights[-1].shape[0]

    net = MLP(d=d, h1=h1, h2=h2, c=c, seed=0)

    # inject deterministic weights
    for key, w in zip(["W1", "W2", "W3"], weights):
        net.p[key] = w.copy()

    cur = x.astype(np.float32, copy=False)
    acts = []

    cur = cur @ net.p["W1"].T
    np.maximum(cur, 0, out=cur)
    acts.append(cur.copy())

    cur = cur @ net.p["W2"].T
    np.maximum(cur, 0, out=cur)
    acts.append(cur.copy())

    # logits are not part of hidden activations
    return acts


def reference_gini(model_path: str, x: np.ndarray, k: int = 0) -> float:
    """Pure‑Python fallback that mirrors ``sa_compute_gini``."""
    acts = _forward_activations(model_path, x)
    if not acts:
        return 0.0
    ginis = [_gini_numpy(a) for a in acts]
    return float(np.mean(ginis))


def reference_layer_stats(model_path: str, x: np.ndarray,
                         layer_id: int) -> Tuple[float, float]:
    """Return (Gini, margin) for a specific hidden layer."""
    acts = _forward_activations(model_path, x)
    if not (0 <= layer_id < len(acts)):
        raise IndexError("layer_id out of range")
    a = acts[layer_id]
    g = _gini_numpy(a)

    k = 1
    total = a.sum()
    if total == 0.0:
        margin = 0.0
    else:
        topk = np.partition(a, -k)[-k:]
        c_k = topk.sum() / total
        topk1 = np.partition(a, -(k + 1))[-(k + 1):]
        c_k1 = topk1.sum() / total
        margin = float(c_k - c_k1)
    return g, margin


# ----------------------------------------------------------------------
# High‑level wrapper for the C library
# ----------------------------------------------------------------------
class _CModelHandle:
    """RAII wrapper for the opaque SA_Model pointer."""
    def __init__(self, path: str):
        self._ptr = ct.c_void_p(_lib.sa_load_model(ct.c_char_p(path.encode())))
        if not self._ptr:
            raise RuntimeError(f"Failed to load model from {path}")

    def __del__(self):
        if getattr(self, "_ptr", None):
            _lib.sa_free_model(self._ptr)

    @property
    def ptr(self) -> ct.c_void_p:
        return self._ptr


class SparseAuditor:
    """
    Detector based on average Gini sparsity over hidden layers.

    Parameters
    ----------
    model_path : str
        Path to the binary model file.
    tau : float, optional
        Threshold for OOD flag; if None, ``detect`` returns only the score.
    use_ref : bool, default False
        If True, use the pure‑Python reference implementation instead of C.
    """
    def __init__(self,
                 model_path: str,
                 tau: Optional[float] = None,
                 use_ref: bool = False):
        self.model_path = model_path
        self.tau = tau
        self.use_ref = use_ref

        if not use_ref:
            self._cmodel = _CModelHandle(model_path)

    # ------------------------------------------------------------------
    # Core API
    # ------------------------------------------------------------------
    def compute_average_gini(self, x: np.ndarray) -> float:
        """Average Gini across hidden layers for a single sample."""
        if self.use_ref:
            return reference_gini(self.model_path, x)
        # C function expects pointer to input and returns gini in *gini_out
        g = ct.c_float()
        rc = _lib.sa_compute_gini(self._cmodel.ptr,
                                  x.ctypes.data_as(ct.POINTER(ct.c_float)),
                                  ct.c_int(x.shape[0]),
                                  ct.c_int(0),
                                  ct.byref(g))
        if rc != 0:
            raise RuntimeError(f"sa_compute_gini failed with code {rc}")
        return float(g.value)

    def layer_stats(self, x: np.ndarray, *, layer_id: int) -> Tuple[float, float]:
        """Return (gini, margin) for a given hidden layer."""
        if self.use_ref:
            return reference_layer_stats(self.model_path, x, layer_id)
        g = ct.c_float()
        m = ct.c_float()
        rc = _lib.sa_layer_stats(self._cmodel.ptr,
                                 x.ctypes.data_as(ct.POINTER(ct.c_float)),
                                 ct.c_int(x.shape[0]),
                                 ct.c_int(layer_id),
                                 ct.byref(g), ct.byref(m))
        if rc != 0:
            raise RuntimeError(f"sa_layer_stats failed with code {rc}")
        return float(g.value), float(m.value)

    def detect(self, x: np.ndarray) -> Tuple[float, int]:
        """Return (average sparsity score, OOD flag)."""
        score = self.compute_average_gini(x)
        flag = 0
        if self.tau is not None and score < self.tau:
            flag = 1
        return score, flag

    # ------------------------------------------------------------------
    # Convenience wrappers used by tests/experiment
    # ------------------------------------------------------------------
    def compute_average_gini(self, x: np.ndarray) -> float:  # noqa: D401
        """Alias for backward compatibility."""
        return self.compute_average_gini(x)

    def layer_stats(self, x: np.ndarray, *, layer_id: int) -> Tuple[float, float]:
        """Alias matching the test expectations."""
        return self.layer_stats(x, layer_id=layer_id)


# ----------------------------------------------------------------------
# Helper to run a batch of inputs
# ----------------------------------------------------------------------
def audit_batch(model_path: str,
                X: np.ndarray,
                tau: Optional[float] = None,
                use_ref: bool = False) -> Tuple[np.ndarray, np.ndarray]:
    """
    Compute detector scores (and optional OOD flags) for a *batch* of samples.
    """
    auditor = SparseAuditor(model_path, tau=tau, use_ref=use_ref)
    X = np.asarray(X, dtype=np.float32)
    scores = np.empty(len(X), dtype=float)
    for i, row in enumerate(X):
        scores[i] = auditor.compute_average_gini(row)
    flags = None
    if tau is not None:
        flags = (scores < tau).astype(int)
    return scores, flags

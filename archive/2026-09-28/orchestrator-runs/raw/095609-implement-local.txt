```python
"""src/algo.py – Sparse Auditor (Algorithm YB‑0003)

Provides:
* CFFI/ctypes bindings to the compiled core library ``libcore.so``.
* A pure‑Python reference implementation that mirrors the C behaviour
  (used for sanity checks in unit tests).
* The high‑level detector class ``SparseAuditor`` exposing a simple API:

    >>> auditor = SparseAuditor("model.bin", tau=0.55)
    >>> score, flag = auditor.detect(x)

The detector returns the average Gini sparsity over all hidden layers and
a binary OOD/adv flag (1 → out‑of‑distribution) based on a single threshold
``tau``.
"""

from __future__ import annotations

import ctypes as ct
import os
from pathlib import Path
from typing import Tuple, Optional

import numpy as np

# ----------------------------------------------------------------------
# Load the shared library (located next to this file)
# ----------------------------------------------------------------------
_lib_path = Path(__file__).with_name("libcore.so")
if not _lib_path.is_file():
    raise FileNotFoundError(f"Compiled core library not found at {_lib_path}")

_lib = ct.CDLL(str(_lib_path))

# ----------------------------------------------------------------------
# C API signatures (as declared in src/core.h)
# ----------------------------------------------------------------------
# SA_Model* sa_load_model(const char *path);
_lib.sa_load_model.argtypes = [ct.c_char_p]
_lib.sa_load_model.restype = ct.c_void_p

# void sa_free_model(SA_Model *model);
_lib.sa_free_model.argtypes = [ct.c_void_p]
_lib.sa_free_model.restype = None

# int sa_compute_gini(const SA_Model *model,
#                     const float *input,
#                     int dim,
#                     int k,
#                     float *gini);
_lib.sa_compute_gini.argtypes = [
    ct.c_void_p,               # model
    ct.POINTER(ct.c_float),   # input pointer
    ct.c_int,                  # dim
    ct.c_int,                  # k (unused)
    ct.POINTER(ct.c_float)    # output gini
]
_lib.sa_compute_gini.restype = ct.c_int

# int sa_layer_stats(const SA_Model *model,
#                    const float *input,
#                    int dim,
#                    int layer_id,
#                    float *gini,
#                    float *margin);
_lib.sa_layer_stats.argtypes = [
    ct.c_void_p,
    ct.POINTER(ct.c_float),
    ct.c_int,
    ct.c_int,
    ct.POINTER(ct.c_float),
    ct.POINTER(ct.c_float)
]
_lib.sa_layer_stats.restype = ct.c_int


# ----------------------------------------------------------------------
# Helper: pure‑Python reference implementation
# ----------------------------------------------------------------------
def _gini_numpy(a: np.ndarray) -> float:
    """Gini coefficient for a non‑negative 1‑D array."""
    if a.size <= 1:
        return 0.0
    s = a.sum()
    if s == 0.0:
        return 0.0
    # sort to make the O(n log n) part explicit (mirrors C qsort)
    sorted_a = np.sort(a)
    n = a.size
    cum = np.cumsum(sorted_a, dtype=np.float64)
    gini = (n + 1 - 2 * np.sum(cum) / s) / n
    return float(gini)


def _forward_activations(model, x: np.ndarray):
    """
    Pure‑Python forward pass using the Python MLP implementation.
    Returns a list of hidden activations (post‑ReLU) for each layer
    except the final output layer.
    """
    # Import lazily to avoid circular imports in tests
    from src.mlp import MLP

    # The binary model file is not needed here; we reconstruct an identical
    # architecture by loading the same weights via the C loader later.
    # For reference we simply instantiate a dummy MLP with matching shape.
    # The test suite provides ``model`` as a path, so we read dimensions from it.
    with open(model, "rb") as f:
        n_layers = int(np.frombuffer(f.read(4), dtype=np.int32)[0])
        dims = []
        for _ in range(n_layers):
            out_dim = int(np.frombuffer(f.read(4), dtype=np.int32)[0])
            in_dim = int(np.frombuffer(f.read(4), dtype=np.int32)[0])
            dims.append((in_dim, out_dim))
            # skip weights
            f.seek(out_dim * in_dim * 4, os.SEEK_CUR)

    # Build a minimal MLP that can perform the forward pass.
    # The actual weight values are irrelevant for reference Gini because the
    # tests compare against the C implementation which uses the same weights.
    mlp = MLP(d=dims[0][0],
              h1=dims[0][1],
              h2=dims[1][1] if len(dims) > 2 else dims[-1][1],
              c=dims[-1][1],
              seed=0)
    # Load weights from the binary file into the Python object
    mlp._load_weights_from_file(model)   # assumes such a helper exists

    acts = []
    cur = x.astype(np.float32, copy=False)
    for W in mlp.weights[:-1]:  # exclude final linear layer
        cur = cur @ W.T
        np.maximum(cur, 0, out=cur)   # ReLU
        acts.append(cur.copy())
    return acts


def reference_gini(model_path: str, x: np.ndarray, k: int = 0) -> float:
    """Pure‑Python fallback that matches ``sa_compute_gini``."""
    acts = _forward_activations(model_path, x)
    if not acts:
        return 0.0
    ginis = [_gini_numpy(a) for a in acts]
    return float(np.mean(ginis))


def reference_layer_stats(model_path: str,
                          x: np.ndarray,
                          layer_id: int) -> Tuple[float, float]:
    """Return (gini, k‑WTA margin) for the requested hidden layer."""
    acts = _forward_activations(model_path, x)
    if layer_id < 0 or layer_id >= len(acts):
        raise IndexError("layer_id out of range")
    a = acts[layer_id]
    g = _gini_numpy(a)

    # k‑WTA margin with k=1 (as in C implementation)
    k = 1
    total = a.sum()
    if total == 0.0:
        margin = 0.0
    else:
        topk = np.partition(a, -k)[-k:]
        c_k = topk.sum() / total
        topk1 = np.partition(a, -(k + 1))[-(k + 1):]
        c_k1 = topk1.sum() / total
        margin = float(c_k - c_k1)
    return g, margin


# ----------------------------------------------------------------------
# High‑level Python wrapper around the C library
# ----------------------------------------------------------------------
class _CModelHandle:
    """RAII wrapper for the opaque SA_Model pointer."""
    def __init__(self, path: str):
        self._ptr = ct.c_void_p(_lib.sa_load_model(ct.c_char_p(path.encode())))
        if not self._ptr:
            raise RuntimeError(f"Failed to load model from {path}")

    def __del__(self):
        if getattr(self, "_ptr", None):
            _lib.sa_free_model(self._ptr)

    @property
    def ptr(self) -> ct.c_void_p:
        return self._ptr


class SparseAuditor:
    """
    Detector based on average Gini sparsity of hidden activations.

    Parameters
    ----------
    model_path : str
        Path to the binary weight file understood by ``src/core.c``.
    tau : float, optional
        Decision threshold on the average Gini.  If omitted,
        ``detect`` returns only the score; the caller can apply any
        custom rule.
    use_ref : bool, default=False
        When True, bypass the C implementation and use the pure‑Python
        reference (useful for debugging or when the shared library is
        unavailable).
    """
    def __init__(self,
                 model_path: str,
                 tau: Optional[float] = None,
                 use_ref: bool = False):
        self.model_path = model_path
        self.tau = tau
        self.use_ref = use_ref

        if not use_ref:
            self._model = _CModelHandle(model_path)

    def _gini_c(self, x: np.ndarray) -> float:
        """Call the C implementation."""
        x_contig = np.ascontiguousarray(x.astype(np.float32))
        out = ct.c_float()
        rc = _lib.sa_compute_gini(
            self._model.ptr,
            x_contig.ctypes.data_as(ct.POINTER(ct.c_float)),
            ct.c_int(x_contig.size),
            ct.c_int(0),               # k unused
            ct.byref(out)
        )
        if rc != 0:
            raise RuntimeError(f"sa_compute_gini returned error code {rc}")
        return float(out.value)

    def _gini_ref(self, x: np.ndarray) -> float:
        """Pure‑Python fallback."""
        return reference_gini(self.model_path, x)

    def compute_average_gini(self, x: np.ndarray) -> float:
        """
        Compute the average Gini coefficient over all hidden layers.

        Returns
        -------
        float
            Average Gini in [0, 1].
        """
        if self.use_ref:
            return self._gini_ref(x)
        else:
            return self._gini_c(x)

    def detect(self,
               x: np.ndarray) -> Tuple[float, Optional[int]]:
        """
        Evaluate a single input.

        Parameters
        ----------
        x : array‑like, shape (d,)
            Input vector.

        Returns
        -------
        score : float
            Average Gini sparsity.
        flag : int or None
            1 if ``score < tau`` (OOD/adv), 0 otherwise. If ``tau`` was not
            supplied at construction, ``flag`` is ``None``.
        """
        x_arr = np.asarray(x, dtype=np.float32).ravel()
        score = self.compute_average_gini(x_arr)
        flag = None
        if self.tau is not None:
            flag = int(score < self.tau)
        return score, flag

    # ------------------------------------------------------------------
    # Additional helper mirroring the C API for per‑layer statistics
    # ------------------------------------------------------------------
    def layer_stats(self,
                    x: np.ndarray,
                    layer_id: int) -> Tuple[float, float]:
        """
        Return (gini, k‑WTA margin) for a specific hidden layer.

        Parameters
        ----------
        x : array‑like, shape (d,)
        layer_id : int
            Zero‑based index among hidden layers (excluding output).

        Returns
        -------
        gini : float
        margin : float
        """
        if self.use_ref:
            return reference_layer_stats(self.model_path,
                                         np.asarray(x, dtype=np.float32), layer_id)

        x_contig = np.ascontiguousarray(np.asarray(x, dtype=np.float32))
        g_out = ct.c_float()
        m_out = ct.c_float()
        rc = _lib.sa_layer_stats(
            self._model.ptr,
            x_contig.ctypes.data_as(ct.POINTER(ct.c_float)),
            ct.c_int(x_contig.size),
            ct.c_int(layer_id),
            ct.byref(g_out),
            ct.byref(m_out)
        )
        if rc != 0:
            raise RuntimeError(f"sa_layer_stats returned error code {rc}")
        return float(g_out.value), float(m_out.value)


# ----------------------------------------------------------------------
# Convenience function used by the test suite and experiment script
# ----------------------------------------------------------------------
def audit_batch(model_path: str,
                X: np.ndarray,
                tau: Optional[float] = None,
                use_ref: bool = False) -> Tuple[np.ndarray, np.ndarray]:
    """
    Compute scores (and optional flags) for a batch of inputs.

    Parameters
    ----------
    model_path : str
        Path to the binary model.
    X : array‑like, shape (N, d)
        Input matrix.
    tau : float, optional
        Threshold for binary flag; if omitted only scores are returned.
    use_ref : bool, default=False
        Force pure‑Python reference implementation.

    Returns
    -------
    scores : ndarray, shape (N,)
    flags  : ndarray of int or None, shape (N,) – ``None`` when tau is None.
    """
    auditor = SparseAuditor(model_path, tau=tau, use_ref=use_ref)
    X_arr = np.asarray(X, dtype=np.float32)
    N = X_arr.shape[0]
    scores = np.empty(N, dtype=float)
    flags = np.empty(N, dtype=int) if tau is not None else None

    for i in range(N):
        s, f = auditor.detect(X_arr[i])
        scores[i] = s
        if flags is not None:
            flags[i] = f
    return scores, flags
```
